Database Schema Definition (MySQL)SQLCREATE DATABASE IF NOT EXISTS sentinel_db;
USE sentinel_db;

CREATE TABLE users (
    id VARCHAR(36) PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    hashed_password VARCHAR(255) NOT NULL,
    role ENUM('USER', 'ADMIN') DEFAULT 'USER',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE analysis_records (
    id VARCHAR(36) PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL,
    file_name VARCHAR(255) NOT NULL,
    file_type ENUM('IMAGE', 'VIDEO') NOT NULL,
    file_url VARCHAR(500) NOT NULL,
    analysis_type ENUM('NSFW', 'SIMILARITY', 'COMBINED') NOT NULL,
    
    -- Safety Metrics (Percentages 0.00 to 100.00)
    score_safe DECIMAL(5, 2) DEFAULT 0.00,
    score_explicit DECIMAL(5, 2) DEFAULT 0.00,
    score_suggestive DECIMAL(5, 2) DEFAULT 0.00,
    score_gore DECIMAL(5, 2) DEFAULT 0.00,
    
    -- OCR Results
    extracted_text TEXT NULL,
    text_flagged BOOLEAN DEFAULT FALSE,
    text_flag_reason VARCHAR(255) NULL,
    
    -- Similarity
    vector_id VARCHAR(64) NULL,
    is_duplicate BOOLEAN DEFAULT FALSE,
    matched_record_id VARCHAR(36) NULL,
    similarity_score DECIMAL(5, 2) DEFAULT 0.00,
    
    -- Explainability
    ai_explanation TEXT NULL,
    overall_status ENUM('SAFE', 'FLAGGED', 'REVIEW') DEFAULT 'SAFE',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE platform_metrics (
    id INT PRIMARY KEY AUTO_INCREMENT,
    metric_key VARCHAR(50) UNIQUE NOT NULL,
    metric_value BIGINT DEFAULT 0,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
